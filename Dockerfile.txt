FROM python:3.11-slim

WORKDIR /app

# Копируем requirements
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Копируем код
COPY main.py .

# Открываем порт
EXPOSE 8080

# Запускаем с оптимизациями для iOS
# --timeout-keep-alive 30 : держим соединения для iOS
# --limit-concurrency 50 : ограничиваем нагрузку
CMD ["python", "-m", "uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8080", "--timeout-keep-alive", "30", "--limit-concurrency", "50"]
